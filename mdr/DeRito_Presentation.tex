	\begin{frame}{Clustering Using K-Means}
		\textbf{K-Means Algorithm:}
		\begin{itemize}
			\item Partitioning data into $k$ clusters.
			\item Optimization based on Euclidean distance.
			\item Silhouette Score for determining optimal $k$:
			\[
			\text{Silhouette Score} = \frac{b - a}{\max(a, b)}
			\]
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Random Forest Classifier for Recommendation}

			\item Voting mechanism for final output.
			\item Robustness against overfitting.
		\end{itemize}

	\begin{frame}{Evaluation Metrics: Classification Report}
		\textbf{Classification Report:}
		\begin{itemize}
			\item Precision ($P$): \( P = \frac{TP}{TP + FP} \)
			\item Recall ($R$): \( R = \frac{TP}{TP + FN} \)
			\item F1-Score: \( F1 = 2 \cdot \frac{P \cdot R}{P + R} \)
		\end{itemize}
		\begin{figure}
			\centering
			\includegraphics[width=0.35\textwidth]{C:/Users/marco/OneDrive/Documents/GitHub/mlops-netflix-branch1/mdr/plots/ClassificationReport.png}
			\caption{Classification Report}
		\end{figure}
		\textbf{Explanation:} This report shows the main classification metrics: precision, recall, and F1-score for each class, helping to assess the performance of the classifier.
	\end{frame}
	
	\begin{frame}{Evaluation Metrics: ROC Curve and AUC}
		\textbf{ROC Curve and AUC:}
		\begin{itemize}
			\item ROC Curve: Plot of True Positive Rate (TPR) vs. False Positive Rate (FPR).
			\item AUC (Area Under the Curve): Measures overall performance. AUC closer to 1 indicates better performance.
		\end{itemize}
		\begin{figure}

		\textbf{Explanation:} The ROC curve is a graphical plot that illustrates the diagnostic ability of a binary classifier. The AUC is a single number summary of the model performance.
